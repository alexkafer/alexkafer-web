# Labs Orb Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the per-section bespoke 3D layouts with a single Fibonacci-sphere "labs orb" driven by a typed lab registry, so adding a new section is one file + one import line. Tone-pass the existing copy in place.

**Architecture:** Single-page narrative, but every section is now a lab module under `src/labs/` exporting `meta` + a default React component. A central `LABS` registry feeds three consumers: the page (renders sections in order), the HUD (shows active lab title + version), and the constellation (positions one star per lab on a Fibonacci sphere, rotates the orb on scroll to bring the active star to a fixed featured anchor). Hero idles with auto-rotation + cursor parallax + drag steering.

**Tech Stack:** Next.js 14 (app router) · React 18 · TypeScript · `@react-three/fiber` + `@react-three/drei` · `three` · `framer-motion` · `tailwindcss`. No test framework in the repo; verification per task is `npx tsc --noEmit` + `npm run lint` + manual scroll-through in `npm run dev`.

**Reference spec:** `docs/superpowers/specs/2026-04-25-labs-orb-design.md`.

```ts
// src/components/hero/sphere-positions.ts
//
// Pure helpers for the labs orb. Stars are distributed on a unit sphere via
// the Fibonacci spiral so that appending a star nudges existing positions
// only slightly (rather than reshuffling them). Edges are picked per-star
// from K nearest neighbours; deduped to undirected pairs.

const GOLDEN_ANGLE = Math.PI * (1 + Math.sqrt(5));

export function fibonacciSphere(count: number): Array<[number, number, number]> {
  if (count <= 0) return [];
  const out: Array<[number, number, number]> = [];
  for (let i = 0; i < count; i++) {
    const phi = Math.acos(1 - (2 * (i + 0.5)) / count);
    const theta = GOLDEN_ANGLE * i;
    const sp = Math.sin(phi);
    out.push([sp * Math.cos(theta), sp * Math.sin(theta), Math.cos(phi)]);
  }
  return out;
}

export function knnEdges(
  points: Array<[number, number, number]>,
  k = 3,
): Array<[number, number]> {
  const n = points.length;
  const seen = new Set<string>();
  const out: Array<[number, number]> = [];
  for (let i = 0; i < n; i++) {
    const dists: Array<[number, number]> = []; // [neighbourIndex, distSq]
    const [ax, ay, az] = points[i];
    for (let j = 0; j < n; j++) {
      if (j === i) continue;
      const [bx, by, bz] = points[j];
      const dx = ax - bx, dy = ay - by, dz = az - bz;
      dists.push([j, dx * dx + dy * dy + dz * dz]);
    }
    dists.sort((a, b) => a[1] - b[1]);
    for (let m = 0; m < Math.min(k, dists.length); m++) {
      const j = dists[m][0];
      const a = Math.min(i, j);
      const b = Math.max(i, j);
      const key = `${a}-${b}`;
      if (seen.has(key)) continue;
      seen.add(key);
      out.push([a, b]);
    }
  }
  return out;
}
```

```ts
// src/components/hero/use-drag-orbit.ts
"use client";

import { useEffect, useRef } from "react";
import * as THREE from "three";

// Returns a quaternion (in a ref) that's additive on top of the orb's
// scroll-driven rotation. Pointer parallax provides a passive tilt; LMB drag
// rotates directly; on release the drag delta decays back to zero so the
// scroll framing wins.
export function useDragOrbit(domEl: HTMLElement | null) {
  const quat = useRef(new THREE.Quaternion());
  const dragging = useRef(false);
  const last = useRef<{ x: number; y: number } | null>(null);
  const dragRot = useRef({ x: 0, y: 0 });
  const cursorTilt = useRef({ x: 0, y: 0 });

  useEffect(() => {
    if (!domEl) return;
    const onDown = (e: PointerEvent) => {
      dragging.current = true;
      last.current = { x: e.clientX, y: e.clientY };
      domEl.setPointerCapture(e.pointerId);
    };
    const onUp = (e: PointerEvent) => {
      dragging.current = false;
      last.current = null;
      try { domEl.releasePointerCapture(e.pointerId); } catch {}
    };
    const onMove = (e: PointerEvent) => {
      const rect = domEl.getBoundingClientRect();
      const nx = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      const ny = ((e.clientY - rect.top) / rect.height) * 2 - 1;
      cursorTilt.current.x = nx * 0.15;
      cursorTilt.current.y = -ny * 0.15;
      if (dragging.current && last.current) {
        const dx = e.clientX - last.current.x;
        const dy = e.clientY - last.current.y;
        dragRot.current.y += dx * 0.005;
        dragRot.current.x += dy * 0.005;
        last.current = { x: e.clientX, y: e.clientY };
      }
    };
    domEl.addEventListener("pointerdown", onDown);
    domEl.addEventListener("pointerup", onUp);
    domEl.addEventListener("pointercancel", onUp);
    domEl.addEventListener("pointermove", onMove);
    return () => {
      domEl.removeEventListener("pointerdown", onDown);
      domEl.removeEventListener("pointerup", onUp);
      domEl.removeEventListener("pointercancel", onUp);
      domEl.removeEventListener("pointermove", onMove);
    };
  }, [domEl]);

  // Caller invokes per-frame to update the additive quaternion.
  // Returns the current quaternion ref for the consumer to multiply in.
  function updateFrame(dt: number) {
    if (!dragging.current) {
      // Decay drag rotation back to zero (~800ms).
      dragRot.current.x *= 0.96;
      dragRot.current.y *= 0.96;
    }
    const rx = dragRot.current.x + cursorTilt.current.y;
    const ry = dragRot.current.y + cursorTilt.current.x;
    const e = new THREE.Euler(rx, ry, 0, "YXZ");
    quat.current.setFromEuler(e);
    return quat.current;
  }

  return { quaternionRef: quat, updateFrame };
}
```

  };
  const out: StarLive[] = [];
  for (let i = 0; i < count; i++) {
    out.push({
      base: new THREE.Vector3(),
      liss: {
        ax: 0.04 + rand() * 0.04,
        ay: 0.04 + rand() * 0.04,
        az: 0.05 + rand() * 0.06,
        px: rand() * Math.PI * 2,
        py: rand() * Math.PI * 2,
        pz: rand() * Math.PI * 2,
        speed: 0.22 + rand() * 0.3,
      },
    });
  }
  return out;
}

function Orb({
  radius,
  baseSize,
  reduced,
  positionsRef,
}: {
  radius: number;
  baseSize: number;
  reduced: boolean;
  positionsRef: React.MutableRefObject<THREE.Vector3[]>;
}) {
  const labs = LABS;
  const total = labs.length;

  // Sphere unit positions, recomputed only when the lab count changes.
  const unitPositions = useMemo(() => fibonacciSphere(total), [total]);
  const stars = useMemo(() => makeLiveStars(total), [total]);

  const meshRefs = useRef<(THREE.Mesh | null)[]>([]);
  const groupRef = useRef<THREE.Group | null>(null);

  // Persistent rotation state; slerps toward target each frame.
  const orbQuat = useRef(new THREE.Quaternion());
  const targetQuat = useRef(new THREE.Quaternion());
  const idleSpinAxis = useMemo(() => new THREE.Vector3(0, 1, 0), []);
  const tmpVec = useMemo(() => new THREE.Vector3(), []);
  const tmpVecB = useMemo(() => new THREE.Vector3(), []);
  const tmpQuatA = useMemo(() => new THREE.Quaternion(), []);
  const tmpQuatB = useMemo(() => new THREE.Quaternion(), []);

  // Map a unit sphere position to world space at the current radius.
  const worldFromUnit = (i: number, q: THREE.Quaternion, out: THREE.Vector3) => {
    const u = unitPositions[i];
    out.set(u[0], u[1], u[2]).multiplyScalar(radius).applyQuaternion(q);
    return out;
  };

  // Compute the rotation that moves star `i`'s sphere position to the anchor.
  const computeFramingQuat = (i: number, target: THREE.Quaternion) => {
    const u = unitPositions[i];
    tmpVec.set(u[0], u[1], u[2]).multiplyScalar(radius); // unrotated star
    tmpVecB.copy(FEATURED_ANCHOR);
    // Quaternion that rotates tmpVec -> tmpVecB
    target.setFromUnitVectors(tmpVec.clone().normalize(), tmpVecB.clone().normalize());
    return target;
  };

  // Initialize positions on first mount.
  useEffect(() => {
    stars.forEach((star, i) => {
      const u = unitPositions[i];
      star.base.set(u[0] * radius, u[1] * radius, u[2] * radius);
    });
    if (positionsRef.current.length !== stars.length) {
      positionsRef.current = stars.map(() => new THREE.Vector3());
    }
  }, [stars, unitPositions, radius, positionsRef]);

  useFrame((state, dt) => {
    const t = state.clock.elapsedTime;
    const sc = getScrollState().current;
    const activeIdx = Math.min(total - 1, Math.max(0, sc.activeIndex));
    const nextIdx = Math.min(total - 1, Math.max(0, sc.nextIndex));

    // Hero (order=0) doesn't feature a star — orb just idles. Otherwise slerp
    // toward the framing rotation for the active star, blending toward `next`
    // in the second half of the section.
    const activeIsHero = activeIdx === 0;
    if (activeIsHero) {
      // Idle slow spin around Y.
      const spin = new THREE.Quaternion().setFromAxisAngle(idleSpinAxis, dt * 0.05);
      targetQuat.current.multiplyQuaternions(spin, targetQuat.current);
    } else {
      computeFramingQuat(activeIdx, tmpQuatA);
      computeFramingQuat(nextIdx, tmpQuatB);
      // Blend A -> B by sc.blend, then this becomes the slerp target.
      tmpQuatA.slerp(tmpQuatB, sc.blend);
      targetQuat.current.copy(tmpQuatA);
    }

    // Slerp current orb rotation toward target.
    orbQuat.current.slerp(targetQuat.current, BASE_LERP);

    // Apply rotation + per-star wobble + featured lift to each mesh.
    stars.forEach((star, i) => {
      const mesh = meshRefs.current[i];
      if (!mesh) return;

      // Rotated sphere position.
      worldFromUnit(i, orbQuat.current, star.base);

      // Featured lift — interpolate between active and next by blend.
      const featuredScale =
        i === activeIdx ? 1 - sc.blend : i === nextIdx ? sc.blend : 0;

      // Wobble (damped during morph).
      const wobble = reduced ? 0 : 1 - sc.blend * 0.5;
      const ox = Math.sin(t * star.liss.speed + star.liss.px) * star.liss.ax * wobble;
      const oy = Math.sin(t * star.liss.speed * 1.3 + star.liss.py) * star.liss.ay * wobble;
      const oz = Math.sin(t * star.liss.speed * 0.9 + star.liss.pz) * star.liss.az * wobble;

      // Apply featured pull toward the anchor.
      const px = star.base.x + (FEATURED_ANCHOR.x - star.base.x) * featuredScale * 0.0;
      const py = star.base.y + (FEATURED_ANCHOR.y - star.base.y) * featuredScale * 0.0;
      const pz = star.base.z + (FEATURED_ANCHOR.z - star.base.z) * featuredScale * 0.0;
      // The framing rotation already places the active star at the anchor —
      // featuredScale only drives visual emphasis (size + colour), not
      // additional translation. The 0.0 multipliers are intentional;
      // remove this block once the rotation framing is verified visually.

      mesh.position.set(px + ox, py + oy, pz + oz);
      positionsRef.current[i].copy(mesh.position);

      // Featured size + colour.
      const labMeta = LABS[i].meta;
      const targetSize =
        baseSize * (1 + featuredScale * (1.0 * (labMeta.starSize ?? 1)));
      mesh.scale.setScalar(targetSize / baseSize);

      const mat = mesh.material as THREE.MeshStandardMaterial;
      const baseCol = new THREE.Color(labMeta.starColor ?? NODE_COLOR);
      const featCol = new THREE.Color(FEATURED_COLOR);
      mat.color.copy(baseCol).lerp(featCol, featuredScale);
      mat.emissive.copy(mat.color);
      mat.emissiveIntensity = 0.6 + featuredScale * 0.6;
    });
  });

  return (
    <group ref={groupRef}>
      {labs.map((lab, i) => (
        <mesh
          key={lab.meta.slug}
          ref={(el) => {
            meshRefs.current[i] = el;
          }}
        >
          <sphereGeometry args={[baseSize, 16, 16]} />
          <meshStandardMaterial
            color={lab.meta.starColor ?? NODE_COLOR}
            emissive={lab.meta.starColor ?? NODE_COLOR}
            emissiveIntensity={0.6}
            roughness={0.4}
            metalness={0.1}
          />
        </mesh>
      ))}
    </group>
  );
}

function OrbLinks({
  positionsRef,
  linkDistance,
}: {
  positionsRef: React.MutableRefObject<THREE.Vector3[]>;
  linkDistance: number;
}) {
  const total = LABS.length;
  // Edges are deterministic for a given lab count: KNN on the Fibonacci sphere.
  const edges = useMemo(() => {
    const pts = fibonacciSphere(total);
    return knnEdges(pts, 3);
  }, [total]);

  const lineRefs = useRef<(THREE.Object3D | null)[]>([]);
  const buf = useRef<Float32Array>(new Float32Array(6));

  useFrame(() => {
    edges.forEach(([i, j], k) => {
      const obj = lineRefs.current[k] as unknown as {
        geometry?: { setPositions?: (arr: ArrayLike<number>) => void };
        material?: { opacity?: number };
      } | null;
      if (!obj?.geometry?.setPositions) return;
      const a = positionsRef.current[i];
      const b = positionsRef.current[j];
      if (!a || !b) return;
      const arr = buf.current;
      arr[0] = a.x; arr[1] = a.y; arr[2] = a.z;
      arr[3] = b.x; arr[4] = b.y; arr[5] = b.z;
      obj.geometry.setPositions(arr);

      const dx = a.x - b.x, dy = a.y - b.y, dz = a.z - b.z;
      const dist = Math.sqrt(dx * dx + dy * dy + dz * dz);
      if (obj.material) {
        const visible = Math.max(0, 1 - dist / linkDistance);
        obj.material.opacity = 0.55 * visible;
      }
    });
  });

  return (
    <group>
      {edges.map(([i, j], k) => (
        <Line
          key={`${i}-${j}`}
          ref={(el) => {
            lineRefs.current[k] = el as unknown as THREE.Object3D | null;
          }}
          points={[[0, 0, 0], [0, 0, 0]]}
          color={NODE_COLOR}
          opacity={0.55}
          transparent
          lineWidth={1.4}
        />
      ))}
    </group>
  );
}

function Scene() {
  const { viewport } = useThree();
  const halfW = viewport.width / 2;
  const halfH = viewport.height / 2;
  const minDim = Math.min(halfW, halfH);

  const radius = minDim * 0.7;
  const baseSize = Math.max(0.06, minDim * 0.025);
  const linkDistance = radius * 1.4;

  const positionsRef = useRef<THREE.Vector3[]>([]);

  const reduced =
    typeof window !== "undefined" &&
    typeof window.matchMedia === "function" &&
    window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  return (
    <>
      <ambientLight intensity={0.4} />
      <pointLight position={[6, 4, 6]} intensity={0.6} color="#7dd3fc" />
      <Starfield />
      <Orb
        radius={radius}
        baseSize={baseSize}
        reduced={reduced}
        positionsRef={positionsRef}
      />
      <OrbLinks positionsRef={positionsRef} linkDistance={linkDistance} />
    </>
  );
}
